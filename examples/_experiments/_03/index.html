DOM++ · Enhanced Document API

Interaction Patterns

A set of patterns for building reactive interactions. The original ElementBuilder and ComponentBuilder engine remains the foundation; document enhancement is an adapter, while JSX is only an alternative syntax.

01 · BASIC DOM

Direct element mutation

Create elements with the builder API, keep references, and update only the nodes that change. No component rerender is involved.

View code (01.0) 
let basicLikes = 0;
let basicCount;
let basicButton;

document.getElementById("basic-app-01.0")
  .setAttributes({ class: "demo-surface basic-surface" })
  .setStyles({ backgroundColor: "#fffebb" })
  .setChildren([
    document.createElement("div")
      .setAttributes({ class: "like-row" })
      .setChildren([
        (basicCount = document.createElement("strong").setText(basicLikes)),
        document.createElement("span").setText(" likes"),
      ]),
    (basicButton = document.createElement("button")
      .setAttributes({ class: "like-button", type: "button" })
      .setText("♥ Like")
      .setEvents({
        click() {
          basicLikes++;
          basicButton.setStyles({
            fontStyle: basicLikes % 2 !== 0 ? "italic" : "normal",
          });
          basicCount.setText(basicLikes);
          if (basicLikes === 10) {
            basicButton.setAttributes({ disabled: true });
          }
        },
      })),
  ]);
View code (01.1) 
let basicLikes = 0;

basicApp = document.createElement(

}

04 · MVVM

ViewModel subscription

The ViewModel owns the count and notifies subscribers. The view rerenders its component in response to updates.

View code 
function LikeViewModel() {
  let likes = 0;
  const listeners = new Set();

  return {
    get likes() { return likes; },
    like() {
      likes++;
      listeners.forEach(listener => listener(likes));
    },
    subscribe(listener) {
      listeners.add(listener);
      return () => listeners.delete(listener);
    }
  };
}

const viewModel = LikeViewModel();
const likeComponent = document.createComponent(() => (
  <section class="demo-surface mvvm-surface">
    <div class="like-row">
      <strong>{viewModel.likes}</strong>
      <span> likes</span>
    </div>
    <button
      class="like-button"
      type="button"
      onClick={() => viewModel.like()}
    >
      ♥ Like
    </button>
  </section>
));

// Mount without using setChildren.
document.createElement(
  document.getElementById("mvvm-app"),
  null,
  likeComponent.render()
);

viewModel.subscribe(() => likeComponent.rerender());
05 · LOCAL STATEFUL IN STATELESS

Local stateful subtree inside a stateless view

A stateless view composes static structure and embeds a document.createComponent subtree. A ViewModel listener triggers rerender only for that local subtree.

View code 
function createViewModel() {
  const listeners = new Set();
  let count = 0;

  return {
    get count() { return count; },
    increment() {
      count++;
      this.notifyListener();
    },
    addListener(listener) {
      listeners.add(listener);
      return () => listeners.delete(listener);
    },
    notifyListener() {
      listeners.forEach((listener) => listener(this));
    },
  };
}

function createView(viewModel) {
  const stateful = document.createComponent(() => (
    <div className="stateful-card">
      <h3>Count: {viewModel.count}</h3>
      <button class="action-button" type="button" onClick={() => viewModel.increment()}>
        UPDATE
      </button>
    </div>
  ));

  const container = (
    <div class="demo-surface lsisv-surface">
      <div class="lsisv-header">
        <h2>Local stateful inside stateless</h2>
        <p>
          The outer container stays static. Only the inner
          <code>createComponent</code> subtree rerenders when the
          ViewModel notifies listeners.
        </p>
      </div>
      {stateful.render()}
    </div>
  );

  viewModel.addListener(() => stateful.rerender());
  return container;
}

const viewModel = createViewModel();
document.createElement(
  document.getElementById("lsisv-app"),
  null,
  createView(viewModel)
);
06 · CONDITIONAL RENDERING

Add and remove a subtree

The component renders a panel only while confirming is true. Returning null removes the panel on the next rerender.

View code 
let confirming = false;

const validator = document.createComponent((rerender) => (
  <section className="demo-surface conditional-surface">
    <h3>Conditional child reconciliation</h3>
    <p>
      The panel is added or removed by rerendering only this component.
      A <code>null</code> branch leaves no node in the virtual tree.
    </p>
    <button
      class="action-button"
      type="button"
      onClick={() => {
        confirming = true;
        rerender();
      }}
    >
      OPEN
    </button>
    {confirming ? (
      <div className="conditional-panel" role="dialog" aria-label="Confirm">
        <h4>Are you sure?</h4>
        <button
          class="action-button"
          type="button"
          onClick={() => {
            confirming = false;
            rerender();
          }}
        >
          CLOSE
        </button>
      </div>
    ) : null}
  </section>
));

document.getElementById("conditional-app").setChildren([
  validator.render(),
]);
07 · KEYED LIST

Stable keys across list changes

Each item passes a stable key to the builder. Reconciliation matches keyed children by key and tag, so append, prepend, reorder, and removal preserve item nodes. Changing an item's tag replaces that node.

View code 
let nextId = 4;
let items = [
  { id: 1, label: "Alpha", tag: "article" },
  { id: 2, label: "Bravo", tag: "article" },
  { id: 3, label: "Charlie", tag: "article" },
];

const keyedList = document.createComponent((rerender) => (
  <section className="demo-surface keyed-surface">
    <h3>Keyed child reconciliation</h3>
    <p>
      Stable keys preserve item nodes when the list order changes.
      Changing an item's tag intentionally replaces its node.
    </p>
    <div className="keyed-controls">
      <button
        class="keyed-button"
        type="button"
        onClick={() => {
          const id = nextId++;
          items = [...items, { id, label: `Item ${id}`, tag: "article" }];
          rerender();
        }}
      >
        APPEND
      </button>
      <button
        class="keyed-button"
        type="button"
        onClick={() => {
          const id = nextId++;
          items = [{ id, label: `Item ${id}`, tag: "article" }, ...items];
          rerender();
        }}
      >
        PREPEND
      </button>
      <button
        class="keyed-button"
        type="button"
        onClick={() => {
          items = [...items].reverse();
          rerender();
        }}
      >
        REVERSE
      </button>
      <button
        class="keyed-button"
        type="button"
        onClick={() => {
          items = items.slice(1);
          rerender();
        }}
      >
        REMOVE FIRST
      </button>
      <button
        class="keyed-button"
        type="button"
        onClick={() => {
          if (items.length) {
            const [first, ...remaining] = items;
            items = [{
              ...first,
              tag: first.tag === "article" ? "p" : "article",
            }, ...remaining];
            rerender();
          }
        }}
      >
        CHANGE FIRST TAG
      </button>
    </div>
    <div className="keyed-list">
      {items.map((item) => {
        const Tag = item.tag;
        return (
          <Tag key={item.id} className="keyed-item">
            {item.label} (key: {item.id})
          </Tag>
        );
      })}
    </div>
  </section>
));

document.getElementById("keyed-list-app").setChildren([
  keyedList.render(),
]);
08 · FRAGMENTS AND ARRAYS

Fragments and arrays

Fragment-producing forms (<>...</>, elementBuilder.fragment, document.createDocumentFragment(...)) return a flat child array that is inlined into the parent. Use <fragment> or document.createElement("fragment") for an explicit wrapper element. Array-returning components are reconciled per slot on rerender.

View code 
let showPanel = false;
let renderCount = 0;

// Array-returning component: reconciled per slot on rerender.
function ArrayChildList({ rerender }) {
  return [
    <div key="a" className="fragment-child fragment-child-a">Array child A</div>,
    <div key="b" className="fragment-child fragment-child-b">
      Array child B (renders: {renderCount})
    </div>,
    <button key="action" class="action-button" type="button"
      onClick={() => rerender(() => renderCount++)}>
      Rerender array slots
    </button>,
  ];
}

// createDocumentFragment(...) returns an inlined child array; as a component
// result its slots are reconciled (null removes a slot).
const inlineFragmentPanel = document.createComponent((rerender) =>
  document.createDocumentFragment(
    <strong>Inline fragment array</strong>,
    <p>Time: {new Date().toLocaleTimeString()}</p>,
    <button class="action-button" type="button"
      onClick={() => rerender(() => (showPanel = true))}>Open panel</button>,
    showPanel ? (
      <div className="conditional-panel" style={{ marginTop: "10px" }}>
        <h4>Conditional slot</h4>
        <button class="action-button" type="button"
          onClick={() => rerender(() => (showPanel = false))}>Close</button>
      </div>
    ) : null
  )
);

const view = document.createComponent((rerender) => (
  <section className="demo-surface fragment-surface">
    {/* <fragment>: explicit wrapper element node */}
    <div className="fragment-box">
      <strong>Parent</strong>
      <fragment>
        <div className="fragment-child fragment-child-a">Wrapped child one</div>
        <div className="fragment-child fragment-child-b">Wrapped child two</div>
      </fragment>
    </div>

    {/* <>...</>: inlined child array, no wrapper */}
    <div className="fragment-box">
      <strong>Parent label</strong>
      <>
        <div className="fragment-child fragment-child-a">Inlined child one</div>
        <div className="fragment-child fragment-child-b">Inlined child two</div>
      </>
    </div>

    {/* Array-returning component, per-slot rerender */}
    <div className="fragment-box">
      <strong>Parent label</strong>
      <ArrayChildList rerender={rerender} />
    </div>

    {/* document.createDocumentFragment(...) child arrays */}
    <div className="fragment-box">
      {document.createDocumentFragment(
        <strong>Static fragment array</strong>,
        <p>Two nodes inlined with no wrapper.</p>
      )}
      {document.createDocumentFragment().setChildren([
        <em key="set-children">Built via .setChildren([...])</em>,
      ])}
      {inlineFragmentPanel.render()}
    </div>
  </section>
));

document.getElementById("fragment-array-app").setChildren([view.render()]);
09 · KEYED LIST IN FRAGMENT

Keyed list inside a real fragment

The list items live inside a real fragment array produced by document.createDocumentFragment(...) (not the <fragment> wrapper element). The fragment inlines its children while stable keys preserve item identity across reorder, insert, and remove.

View code 
let nextId = 4;
let items = [
  { id: 1, label: "Alpha" },
  { id: 2, label: "Bravo" },
  { id: 3, label: "Charlie" },
];

const keyedFragment = document.createComponent((rerender) => (
  <section className="demo-surface keyed-surface">
    <h3>Keyed list inside a fragment</h3>
    <p>
      Items are produced inside a real fragment array via
      <code>document.createDocumentFragment(...)</code> (not the
      <code>&lt;fragment&gt;</code> wrapper). The fragment inlines
      its children while stable keys preserve item identity.
    </p>
    <div className="keyed-controls">
      <button class="keyed-button" type="button"
        onClick={() => { const id = nextId++; items = [...items, { id, label: `Item ${id}` }]; rerender(); }}>
        APPEND
      </button>
      <button class="keyed-button" type="button"
        onClick={() => { const id = nextId++; items = [{ id, label: `Item ${id}` }, ...items]; rerender(); }}>
        PREPEND
      </button>
      <button class="keyed-button" type="button"
        onClick={() => { items = [...items].reverse(); rerender(); }}>
        REVERSE
      </button>
      <button class="keyed-button" type="button"
        onClick={() => { if (items.length) { items = items.slice(1); rerender(); } }}>
        REMOVE FIRST
      </button>
    </div>
    <div className="keyed-list">
      {document.createDocumentFragment(
        ...items.map((item) => (
          <div key={item.id} className="keyed-item">
            {item.label} (key: {item.id})
          </div>
        ))
      )}
    </div>
  </section>
));

10 · SERVER DOM ADOPTION

Attach to existing (server-rendered) markup

The <ul> below is present in the HTML before any script runs (as server-rendered output would be). The engine adopts that existing node and reconciles a keyed descriptor tree against it, preserving node identity instead of recreating the DOM.

Server-rendered list

This list is plain HTML in the document. Adoption attaches a virtual tree to it.

  • JavaScript will adopt it
  • This list was rendered as HTML
  • Hello from the server

Waiting for adoption...

View code 
// Data embedded by the server alongside the rendered HTML.
const serverMessages = JSON.parse(
  document.getElementById("adoption-data").textContent
);

const listNode = document.getElementById("adoption-list");

// Adopt the existing <ul> and reconcile a keyed descriptor tree
// against its current children instead of recreating them.
const listBuilder = document.createElement(
  listNode,
  null,
  ...[...serverMessages]
    .reverse()
    .map((msg) => <li key={msg.id}>{msg.text}</li>)
);

// listBuilder.node === listNode && node identity is preserved
document.getElementById("adoption-status").setText(
  "Adopted existing <ul>: " + (listBuilder.node === listNode) +
  " · reconciled items: " + listBuilder.struct.children.length
);
11 · REACTIVE SERVER MESSENGER

Adopt server markup, then rerender reactively

The container is rendered in HTML up front. A component adopts that existing root with document.createElement(existingNode, ...) and rerenders a keyed message list on Send, keeping the same root node attached.

Reactive server messenger

Server-rendered placeholder. The component will adopt and control this root.

  • Welcome! This came from the server.
  • Type below and press Send.

Waiting for component to attach...

View code 
// Data embedded by the server alongside the rendered HTML.
const messages = JSON.parse(
  document.getElementById("messenger-data").textContent
);
let nextId = Math.max(...messages.map((m) => m.id)) + 1;
const rootNode = document.getElementById("messenger-app");

const messenger = document.createComponent((rerender) =>
  document.createElement(
    rootNode,
    { class: "demo-surface messenger-surface" },
    <h3>Reactive server messenger</h3>,
    <p>Adopts the pre-rendered root and rerenders a keyed list.</p>,
    <ul class="server-list">
      {[...messages].reverse().map((msg) => (
        <li key={msg.id}>{msg.text}</li>
      ))}
    </ul>,
    <div class="message-form">
      <textarea class="message-input" placeholder="Write a message..."></textarea>
      <button class="action-button" type="button" onClick={() => {
        const input = rootNode.querySelector("textarea");
        const text = input.value.trim();
        if (!text) return;
        messages.push({ id: nextId++, text });
        input.value = "";
        rerender();
      }}>Send</button>
    </div>,
    <p id="messenger-status" class="status-note">
      {"Adopted root node · messages: " + messages.length}
    </p>
  )
);

messenger.render();
